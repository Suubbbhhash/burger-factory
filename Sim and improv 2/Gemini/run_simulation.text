function randomItem(arr) {
  return arr[Math.floor(Math.random() * arr.length)];
}

function randomPhone() {
  const prefixes = ['98', '97', '99', '96', '91', '88', '87', '77'];
  const p = randomItem(prefixes);
  let rest = '';
  for (let i = 0; i < 8; i++) rest += Math.floor(Math.random() * 10);
  return `+91 ${p}${rest.slice(0, 3)} ${rest.slice(3)}`;
}

function getISTISODate(offsetDays = 0) {
  const istMs = Date.now() + 19800000 + offsetDays * 86400000;
  return new Date(istMs).toISOString().slice(0, 10);
}

function timeToMinutes(time) {
  if (!time || typeof time !== 'string' || !time.includes(':')) return NaN;
  const [h, m] = time.split(':').map(Number);
  return h * 60 + m;
}

// ---------- Self-Contained Embedded Server Fallback ----------
async function ensureServerRunning() {
  const probe = await apiRequest('GET', '/api/v1/reservations');
  if (probe.status !== 0) {
    return { embeddedServer: null };
  }

  console.log(`[i] No external server detected on ${BASE_URL}; booting embedded LAN API server...`);
  const middlewares = [];
  const plugin = burgerFactoryApiPlugin();
  plugin.configureServer({
    middlewares: {
      use(fn) {
        middlewares.push(fn);
      }
    }
  });

  const server = http.createServer((req, res) => {
    let idx = 0;
    const next = () => {
      const fn = middlewares[idx++];
      if (fn) {
        fn(req, res, next);
      } else {
        res.writeHead(404, { 'Content-Type': 'application/json' });
        res.end(JSON.stringify({ error: 'Not found' }));
      }
    };
    next();
  });

  const urlObj = new URL(BASE_URL);
  const port = Number(urlObj.port) || 3000;
  await new Promise((resolve, reject) => {
    server.once('error', reject);
    server.listen(port, resolve);
  });
  console.log(`[✓] Embedded LAN API server listening at ${BASE_URL}`);
  return { embeddedServer: server };
}

// ---------- HTTP Helper with Latency Tracking ----------
async function apiRequest(method, endpoint, body = null) {
  const t0 = performance.now();
  telemetry.totalHttpRequests++;
  try {
    const opts = {
      method,
      headers: { 'Content-Type': 'application/json' }
    };
    if (body !== null) {
      opts.body = typeof body === 'string' ? body : JSON.stringify(body);
    }
    const res = await fetch(`${BASE_URL}${endpoint}`, opts);
    const elapsed = performance.now() - t0;
    telemetry.latenciesMs.push(elapsed);

    let data = null;
    const text = await res.text();
    try {
      data = JSON.parse(text);
    } catch {
      data = { rawText: text };
    }
    return { ok: res.ok, status: res.status, data, elapsed };
  } catch (err) {
    const elapsed = performance.now() - t0;
    telemetry.latenciesMs.push(elapsed);
    return { ok: false, status: 0, error: err.message, data: null, elapsed };
  }
}

// ---------- SSE Stream Listener ----------
function connectSseListener() {
  return new Promise((resolve) => {
    const url = new URL('/api/v1/events', BASE_URL);
    const req = http.get(url, (res) => {
      res.on('data', (chunk) => {
        const str = chunk.toString();
        const matches = str.match(/event:\s*(\S+)/g);
        if (matches) {
          telemetry.sseEventsReceived += matches.length;
        }
      });
      resolve({
        close: () => {
          try { req.destroy(); } catch {}
        }
      });
    });
    req.on('error', () => {
      resolve({ close: () => {} });
    });
  });
}

// ---------- Client-Mirror Availability Check ----------
function findBestTableClientSide(existingReservations, locationId, date, time, guests, seatingPref = 'any') {
  const tables = locationTables[locationId];
  if (!tables) return null;

  const locObj = locations.find(l => l.id === locationId);
  if (locObj?.lastSlot && timeToMinutes(time) > timeToMinutes(locObj.lastSlot)) {
    return null;
  }

  let allTables = [];
  for (const [zone, zoneTables] of Object.entries(tables)) {
    for (const t of zoneTables) {
      allTables.push({ ...t, zone });
    }
  }

  if (seatingPref && seatingPref !== 'any') {
    const prefMap = {
      indoor: ['indoor'],
      outdoor: ['outdoor', 'beachside', 'rooftop'],
      beachside: ['beachside', 'outdoor'],
      rooftop: ['rooftop', 'outdoor']
    };
    const allowed = prefMap[seatingPref] || [seatingPref];
    const filtered = allTables.filter(t => allowed.includes(t.zone));
    if (filtered.length > 0) allTables = filtered;
  }

  allTables = allTables
    .filter(t => t.capacity >= guests)
    .sort((a, b) => a.capacity - b.capacity);

  const reqMin = timeToMinutes(time);

  for (const table of allTables) {
    const hasCollision = existingReservations.some(r => {
      if (['cancelled', 'no-show', 'completed'].includes(r.status)) return false;
      if (r.locationId !== locationId || r.date !== date || r.tableId !== table.id) return false;
      const bMin = timeToMinutes(r.time);
      return Math.abs(reqMin - bMin) < TURN_MINUTES;
    });
    if (!hasCollision) return table;
  }
  return null;
}

const BOOKING_CHARS = '23456789ABCDEFGHJKLMNPQRSTUVWXYZ';
function genCode() {
  let id = 'BF-';
  for (let i = 0; i < 6; i++) {
    id += BOOKING_CHARS.charAt(Math.floor(Math.random() * BOOKING_CHARS.length));
  }
  return id;
}

  const activeRecords = allRecords.filter(r => ['confirmed', 'seated'].includes(r.status));

      if (a.locationId === b.locationId && a.date === b.date && a.tableId === b.tableId) {

        const minB = timeToMinutes(b.time);

          collisions.push({
            locationId: a.locationId,
            date: a.date,
            tableId: a.tableId,
            bookingA: `${a.code} (${a.name} @ ${a.time}, ${a.status})`,
            bookingB: `${b.code} (${b.name} @ ${b.time}, ${b.status})`
          });
        }

    }

      ).join('\n'),
      'Ensure allocateTableAtomic runs inside withMutex in vite.config.js.'

    telemetry.invariantsPassed++;

  // Invariant 2: Zero Table Capacity Violations
  const capViolations = allRecords.filter(r => {
    const tbl = tableLookup.get(`${r.locationId}:${r.tableId}`);
    return !tbl || r.guests > tbl.capacity || r.guests < 1;
  });
  if (capViolations.length > 0) {
    logError(
      'CAPACITY_ENFORCEMENT',
      'HIGH',
      'ERR-CAP-002',
      `Detected ${capViolations.length} reservations exceeding assigned table capacity`,
      capViolations.slice(0, 5).map(v => `  - ${v.code}: ${v.guests} pax on ${v.tableId}`).join('\n'),
      'Verify table.capacity >= guests in allocateTableAtomic.'
    );
  } else {
    telemetry.invariantsPassed++;
  }

  // Invariant 3: Zero Duplicate Codes & Zero Ghost Confirmations
  const seenCodes = new Set();
  const duplicateCodes = [];
  for (const r of allRecords) {
    if (seenCodes.has(r.code)) duplicateCodes.push(r.code);
    seenCodes.add(r.code);
  }
  const missingFromLedger = [...telemetry.confirmedCodesLedger].filter(c => !seenCodes.has(c));
  if (duplicateCodes.length > 0 || missingFromLedger.length > 0) {
    logError(
      'LEDGER_INTEGRITY',
      'CRITICAL',
      'ERR-GHOST-001',
      `Ledger mismatch: ${duplicateCodes.length} duplicate codes, ${missingFromLedger.length} ghost confirmations`,
      `Duplicates: ${duplicateCodes.join(', ') || 'none'} | Missing: ${missingFromLedger.slice(0, 5).join(', ') || 'none'}`,
      'Ensure atomic file writes and server-acknowledged reservation codes.'
    );
  } else {
    telemetry.invariantsPassed++;
  }

  // Invariant 4: Lifecycle Timestamp Integrity
  const missingTimestamps = allRecords.filter(r =>
    (r.status === 'seated' && !r.seatedAt) ||
    (r.status === 'completed' && !r.completedAt) ||
    (r.status === 'cancelled' && !r.cancelledAt) ||
    (r.status === 'no-show' && !r.noShowAt)
  );
  if (missingTimestamps.length > 0) {
    logError(
      'LIFECYCLE_TIMESTAMPS',
      'MEDIUM',
      'ERR-FSM-004',
      `Detected ${missingTimestamps.length} transitioned records missing lifecycle timestamps`,
      missingTimestamps.slice(0, 5).map(r => `  - ${r.code} (${r.status})`).join('\n'),
      'Stamp seatedAt, completedAt, cancelledAt, and noShowAt on status transitions.'
    );
  } else {
    telemetry.invariantsPassed++;
  }

  // Invariant 5: Outlet Metadata vs Physical Table Map Consistency
  const dataMismatches = [];
  for (const loc of locations) {

    let tableCount = 0;
    let seatSum = 0;
    for (const list of Object.values(zones)) {
      tableCount += list.length;
      for (const t of list) seatSum += t.capacity;
    }
    if (loc.tables !== tableCount || loc.capacity !== seatSum) {
      dataMismatches.push(`${loc.id}: declared ${loc.tables} tables/${loc.capacity} seats vs actual ${tableCount} tables/${seatSum} seats`);
    }
  }
  if (dataMismatches.length > 0) {
    logError(
      'DATA_CONSISTENCY',
      'MEDIUM',
      'ERR-DATA-001',
      'Mismatch between locations metadata and locationTables seat sums in data.js',
      dataMismatches.join('\n'),
      'Align locations[i].tables and locations[i].capacity with locationTables in src/js/data.js.'
    );
  } else {
    telemetry.invariantsPassed++;
  }

  // Invariant 6: Real-Time SSE Broadcast Delivery
  if (telemetry.sseEventsReceived === 0) {
    logError(
      'REALTIME_SSE',
      'HIGH',
      'ERR-SSE-001',
      'Zero SSE broadcast events received during simulation run',

  } else {
    telemetry.invariantsPassed++;

        'MEDIUM',

        'Remove `value="burger123"` from `#staff-gate-pass` in manager.html.'

    }

        logError(
          'TIMEZONE_DRIFT',
          'MEDIUM',
          'ERR-TZ-001',

        );
      }
    }
  }
  if (tzClean) telemetry.invariantsPassed++;

  if (fs.existsSync(managerJsPath)) {
    const managerJs = fs.readFileSync(managerJsPath, 'utf8');

      floorClean = false;
      logError(
        'FLOOR_MAP_LOGIC',
        'MEDIUM',
        'ERR-FLOOR-001',
        'Floor Plan compares future/past service dates against today\'s live wall-clock hour',
        'renderFloorCanvas() filters by currentMinutes without checking currentDate === todayStr.',
        'Only compare against currentMinutes when currentDate === todayStr.'
      );
    }
  }
  if (floorClean) telemetry.invariantsPassed++;

  // Invariant 10: Frontend XSS Escaping & Workspace Sync Hygiene
  let xssFrontendClean = true;
  for (const relFile of ['src/js/booking.js', 'src/js/main.js', 'src/js/manager.js', 'src/js/thermalTicket.js']) {
    const fullPath = path.join(ROOT_DIR, relFile);
    if (fs.existsSync(fullPath)) {
      const content = fs.readFileSync(fullPath, 'utf8');
      if (!content.includes('escapeHtml')) {
        xssFrontendClean = false;
        logError(
          'SECURITY_XSS',
          'HIGH',
          'ERR-SEC-003',
          `Missing escapeHtml() protection in ${relFile}`,
          `${relFile} renders reservation fields into innerHTML without escapeHtml().`,
          `Import and apply escapeHtml() in ${relFile}.`
        );
      }
    }
  }
  const webIndexHtml = path.join(ROOT_DIR, 'apps', 'web', 'index.html');
  const rootIndexHtml = path.join(ROOT_DIR, 'index.html');
  if (fs.existsSync(webIndexHtml) && fs.existsSync(rootIndexHtml)) {
    const wSize = fs.statSync(webIndexHtml).size;
    const rSize = fs.statSync(rootIndexHtml).size;
    if (wSize !== rSize) {
      logWarning(
        'WARN-SYNC-001',
        `Workspace divergence detected: /index.html (${rSize} B) differs from /apps/web/index.html (${wSize} B)`,
        'Keep /apps/web synchronized with root files.'
      );
    }
  }
  if (xssFrontendClean) telemetry.invariantsPassed++;

  sseClient.close();
  telemetry.endTime = new Date().toISOString();

  // Restore clean initial seed database after simulation audit
  if (fs.existsSync(BACKUP_DB_FILE)) {
    fs.copyFileSync(BACKUP_DB_FILE, DB_FILE);
    fs.unlinkSync(BACKUP_DB_FILE);

  } else {

    console.log(`[✓] Reset reservations database to clean initial seed after simulation audit.`);
  }

  if (embeddedServer) {
    await new Promise(resolve => embeddedServer.close(resolve));
  }

  writeSimulationReport(allRecords.length, collisions.length);
